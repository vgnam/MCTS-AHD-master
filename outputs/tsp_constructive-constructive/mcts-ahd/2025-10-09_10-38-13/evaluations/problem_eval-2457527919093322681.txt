def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node
    min_score = float('inf')
    next_node = None
    remaining_proportion = len(unvisited_nodes) / len(distance_matrix)
    for node in unvisited_nodes:
        distance_to_current = distance_matrix[current_node][node]
        distance_to_dest = distance_matrix[node][destination_node]
        remaining_nodes = unvisited_nodes - {node}
        if remaining_nodes:
            farthest_node = max(remaining_nodes, key=lambda n: distance_matrix[node][n])
            max_remaining_distance = distance_matrix[node][farthest_node]
            centrality = sum(distance_matrix[node][n] for n in remaining_nodes) / len(remaining_nodes)
            avg_remaining_distance = centrality
        else:
            max_remaining_distance = 0
            centrality = 0
            avg_remaining_distance = 0
        local_weight = 0.6 - 0.2 * remaining_proportion
        global_weight = 0.2 + 0.2 * remaining_proportion
        centrality_weight = 0.1 + 0.1 * remaining_proportion
        penalty_weight = 0.1 + 0.1 * remaining_proportion
        score = (local_weight * distance_to_current) + (global_weight * distance_to_dest) + (centrality_weight * centrality) - (penalty_weight * max_remaining_distance) - (0.05 * avg_remaining_distance)
        if score < min_score:
            min_score = score
            next_node = node
    return next_node
