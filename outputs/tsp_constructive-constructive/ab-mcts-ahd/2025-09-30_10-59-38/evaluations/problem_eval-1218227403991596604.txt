def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    def heuristic_score(node):
        if node == destination_node:
            return 0
        remaining_nodes = len(unvisited_nodes)
        total_nodes = len(distance_matrix)
        local_weight = 1.0 - (remaining_nodes / total_nodes) * 0.5
        global_weight = 1.0 - local_weight
        return (local_weight * distance_matrix[current_node][node] +
                global_weight * distance_matrix[node][destination_node])

    next_node = min(unvisited_nodes, key=lambda node: heuristic_score(node))
    return next_node
