def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    def heuristic_score(node):
        if node == destination_node:
            return 0
        local_proximity = distance_matrix[current_node][node]
        global_efficiency = distance_matrix[node][destination_node]

        if not unvisited_nodes:
            return local_proximity

        unvisited_distances = [distance_matrix[current_node][n] for n in unvisited_nodes]
        avg_unvisited = sum(unvisited_distances) / len(unvisited_distances)
        var_unvisited = sum((d - avg_unvisited) ** 2 for d in unvisited_distances) / len(unvisited_distances)

        # Dynamic weight based on variance and sigmoid normalization
        weight = 1 / (1 + math.exp(-(local_proximity - avg_unvisited) / (var_unvisited + 1e-6)))

        # Penalize nodes that disrupt path smoothness
        if len(unvisited_nodes) > 1:
            other_nodes = [n for n in unvisited_nodes if n != node]
            avg_angle_penalty = sum(
                abs(math.atan2(distance_matrix[node][n] - distance_matrix[current_node][node],
                              distance_matrix[current_node][n]) - math.pi)
                for n in other_nodes
            ) / len(other_nodes)
            smoothness_penalty = avg_angle_penalty * (local_proximity / avg_unvisited)
        else:
            smoothness_penalty = 0

        return local_proximity + weight * global_efficiency + smoothness_penalty

    next_node = min(unvisited_nodes, key=lambda node: heuristic_score(node))
    return next_node
