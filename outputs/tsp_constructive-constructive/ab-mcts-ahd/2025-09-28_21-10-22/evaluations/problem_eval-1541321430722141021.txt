def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    next_node = None
    best_score = float('inf')
    remaining_nodes = len(unvisited_nodes)

    for node in unvisited_nodes:
        direct_distance = distance_matrix[current_node][node]
        potential_reduction = sum(distance_matrix[node][v] for v in unvisited_nodes if v != node)
        weight = 1 / (1 + math.exp(-(remaining_nodes / len(distance_matrix))))
        score = (1 - weight) * direct_distance + weight * potential_reduction

        if score < best_score:
            best_score = score
            next_node = node

    return next_node
