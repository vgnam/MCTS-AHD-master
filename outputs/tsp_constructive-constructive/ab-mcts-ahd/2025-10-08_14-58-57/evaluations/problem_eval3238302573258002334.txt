def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    remaining_nodes = len(unvisited_nodes)
    avg_distance = sum(distance_matrix[current_node][node] for node in unvisited_nodes) / remaining_nodes
    weight = 1.0 / (1.0 + 0.3 * (remaining_nodes / len(distance_matrix)))
    next_node = min(unvisited_nodes, key=lambda node: distance_matrix[current_node][node] + weight * distance_matrix[node][destination_node] + 0.1 * abs(distance_matrix[current_node][node] - avg_distance))
    return next_node
