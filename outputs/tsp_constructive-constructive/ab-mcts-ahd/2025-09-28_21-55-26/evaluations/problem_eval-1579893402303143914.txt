def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node
    if len(unvisited_nodes) == 1:
        return unvisited_nodes.pop()

    # Phase 1: Compute nearest-neighbor candidate
    nearest_node = min(unvisited_nodes, key=lambda node: distance_matrix[current_node][node])

    # Phase 2: Compute weighted insertion cost with centrality and detour minimization
    def insertion_cost(node):
        if len(unvisited_nodes) == 1:
            return distance_matrix[current_node][node] + distance_matrix[node][destination_node]

        # Centrality metric: average distance to other nodes
        centrality = sum(distance_matrix[node][n] for n in unvisited_nodes if n != node) / (len(unvisited_nodes) - 1)

        # Greedy insertion cost with detour minimization
        remaining_nodes = unvisited_nodes - {node}
        if not remaining_nodes:
            return distance_matrix[current_node][node] + distance_matrix[node][destination_node]

        next_nearest = min(remaining_nodes, key=lambda n: distance_matrix[node][n])
        base_cost = (distance_matrix[current_node][node] +
                    distance_matrix[node][next_nearest] -
                    distance_matrix[current_node][next_nearest])

        # Weighted cost combining base insertion cost, centrality, and detour minimization
        return base_cost * 0.6 + centrality * 0.3 + distance_matrix[node][next_nearest] * 0.1

    # Select node with minimal weighted cost
    next_node = min(unvisited_nodes, key=insertion_cost)
    return next_node
