def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node
    if len(unvisited_nodes) == 1:
        return unvisited_nodes.pop()

    # Phase 1: Select the nearest unvisited node (local proximity)
    nearest_node = min(unvisited_nodes, key=lambda node: distance_matrix[current_node][node])

    # Phase 2: Evaluate insertion cost with dynamic weight factor
    total_unvisited = len(unvisited_nodes)
    weight_factor = 1 - (0.5 * (total_unvisited / (total_unvisited + 1)))

    def evaluate_node(node):
        immediate_distance = distance_matrix[current_node][node]
        if len(unvisited_nodes) == 1:
            return immediate_distance

        remaining_nodes = unvisited_nodes - {node}
        if not remaining_nodes:
            return immediate_distance

        # Insertion cost: cost of inserting node between current and next nearest
        next_nearest = min(remaining_nodes, key=lambda n: distance_matrix[node][n])
        insertion_cost = (distance_matrix[current_node][node] +
                          distance_matrix[node][next_nearest] -
                          distance_matrix[current_node][next_nearest])

        # Combine with dynamic weight factor
        return (weight_factor * immediate_distance) + ((1 - weight_factor) * insertion_cost)

    next_node = min(unvisited_nodes, key=evaluate_node)
    return next_node
