def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node
    if len(unvisited_nodes) == 1:
        return unvisited_nodes.pop()

    total_unvisited = len(unvisited_nodes)
    weight_factor = total_unvisited / (total_unvisited + 1)

    def evaluate_node(node):
        immediate_distance = distance_matrix[current_node][node]
        remaining_nodes = unvisited_nodes - {node}
        if not remaining_nodes:
            return immediate_distance
        avg_remaining_distance = sum(distance_matrix[node][other] for other in remaining_nodes) / len(remaining_nodes)
        return (weight_factor * immediate_distance) + ((1 - weight_factor) * avg_remaining_distance)

    next_node = min(unvisited_nodes, key=evaluate_node)
    return next_node
