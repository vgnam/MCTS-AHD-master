def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node
    if len(unvisited_nodes) == 1:
        return unvisited_nodes.pop()

    # Phase 1: Calculate proximity and regret for all unvisited nodes
    proximity = {node: distance_matrix[current_node][node] for node in unvisited_nodes}
    regret = {}
    for node in unvisited_nodes:
        remaining_nodes = unvisited_nodes - {node}
        if not remaining_nodes:
            regret[node] = 0
        else:
            nearest_after = min(remaining_nodes, key=lambda n: distance_matrix[node][n])
            regret[node] = (distance_matrix[node][nearest_after] -
                           min(distance_matrix[node][n] for n in remaining_nodes if n != nearest_after))

    # Phase 2: Dynamic weighting based on remaining nodes and regret
    beta = 0.5 * (len(unvisited_nodes) / (len(unvisited_nodes) + 1))  # Exploration-exploitation balance
    weighted_scores = {
        node: beta * proximity[node] + (1 - beta) * regret[node] +
              0.2 * (distance_matrix[node][destination_node] if len(unvisited_nodes) > 2 else 0)
        for node in unvisited_nodes
    }

    # Select node with minimal weighted score
    next_node = min(weighted_scores, key=weighted_scores.get)
    return next_node
