def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    remaining_nodes = len(unvisited_nodes)
    progress = 1.0 - (remaining_nodes / (remaining_nodes + 1))
    weight_factor = (remaining_nodes ** 2) / (len(distance_matrix) ** 2) * (1.0 - progress)
    next_node = min(unvisited_nodes, key=lambda node: (weight_factor * distance_matrix[current_node][node] + (1 - weight_factor) * (distance_matrix[current_node][destination_node] - distance_matrix[node][destination_node])))
    return next_node
