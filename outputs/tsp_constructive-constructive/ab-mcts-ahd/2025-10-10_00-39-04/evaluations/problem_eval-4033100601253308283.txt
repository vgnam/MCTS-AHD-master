def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    remaining_nodes = len(unvisited_nodes)
    weight_immediate_cost = 0.65 - 0.15 * (remaining_nodes / (remaining_nodes + 1))
    weight_heuristic_term = 0.30 + 0.15 * (remaining_nodes / (remaining_nodes + 1))
    weight_novelty_factor = 0.05 * (1 - (remaining_nodes / (remaining_nodes + 1)))

    def evaluate_node(node):
        immediate_cost = distance_matrix[current_node][node]
        heuristic_term = distance_matrix[current_node][destination_node] - distance_matrix[node][destination_node]
        novelty_factor = sum(distance_matrix[node][n] for n in unvisited_nodes) / len(unvisited_nodes)
        return (weight_immediate_cost * immediate_cost + 
                weight_heuristic_term * heuristic_term - 
                weight_novelty_factor * novelty_factor)

    next_node = min(unvisited_nodes, key=lambda node: evaluate_node(node))
    return next_node
