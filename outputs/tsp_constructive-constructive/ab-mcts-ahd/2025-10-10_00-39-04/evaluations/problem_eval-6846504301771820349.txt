def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix, visited_nodes=set()):
    remaining_nodes = len(unvisited_nodes)
    weight_immediate_cost = 0.6 - 0.1 * (remaining_nodes / (remaining_nodes + 1))
    weight_heuristic_term = 0.4 + 0.1 * (remaining_nodes / (remaining_nodes + 1))
    weight_proximity_bonus = 0.2 * (remaining_nodes / (remaining_nodes + 1))

    def evaluate_node(node):
        immediate_cost = distance_matrix[current_node][node]
        heuristic_term = distance_matrix[current_node][destination_node] - distance_matrix[node][destination_node]
        proximity_bonus = sum(distance_matrix[node][v] for v in visited_nodes) if visited_nodes else 0
        return (weight_immediate_cost * immediate_cost + 
                weight_heuristic_term * heuristic_term - 
                weight_proximity_bonus * proximity_bonus)

    next_node = min(unvisited_nodes, key=lambda node: evaluate_node(node))
    return next_node
