def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    total_nodes = len(distance_matrix)
    remaining_nodes = len(unvisited_nodes)
    visited_ratio = (total_nodes - remaining_nodes) / total_nodes
    immediate_weight = 0.7 * (1 - visited_ratio) + 0.3 * visited_ratio
    destination_weight = (remaining_nodes ** 2) / (total_nodes ** 2)
    future_reduction_weight = 0.3 * (1 - destination_weight)

    def evaluate_node(node):
        immediate_cost = immediate_weight * distance_matrix[current_node][node]
        destination_cost = destination_weight * distance_matrix[node][destination_node]
        future_reduction_cost = future_reduction_weight * (distance_matrix[current_node][destination_node] - distance_matrix[node][destination_node])
        return immediate_cost + destination_cost + future_reduction_cost

    next_node = min(unvisited_nodes, key=lambda node: max(evaluate_node(node), distance_matrix[node][destination_node]))
    return next_node
