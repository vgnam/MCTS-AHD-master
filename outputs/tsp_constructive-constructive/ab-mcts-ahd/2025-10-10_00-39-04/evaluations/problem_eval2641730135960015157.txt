def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    remaining_nodes = len(unvisited_nodes)
    temperature = 0.5 + 0.5 * (remaining_nodes / (remaining_nodes + 1))
    weight_immediate_cost = 0.60 + 0.10 * (remaining_nodes / (remaining_nodes + 1))
    weight_heuristic_term = 0.30 - 0.10 * (remaining_nodes / (remaining_nodes + 1))
    weight_diversity_factor = 0.10 + 0.10 * (remaining_nodes / (remaining_nodes + 1))

    def evaluate_node(node):
        immediate_cost = distance_matrix[current_node][node]
        heuristic_term = distance_matrix[current_node][destination_node] - distance_matrix[node][destination_node]
        centrality = sum(distance_matrix[node][n] for n in unvisited_nodes) / len(unvisited_nodes)
        diversity_factor = (centrality - sum(distance_matrix[node][n] for n in unvisited_nodes) / len(unvisited_nodes)) * temperature

        return (weight_immediate_cost * immediate_cost +
                weight_heuristic_term * heuristic_term -
                weight_diversity_factor * diversity_factor)

    next_node = min(unvisited_nodes, key=lambda node: evaluate_node(node))
    return next_node
