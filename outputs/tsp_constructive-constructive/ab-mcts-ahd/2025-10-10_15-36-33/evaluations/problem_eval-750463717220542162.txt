def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    remaining_nodes = len(unvisited_nodes)
    weight_current = 0.7 if remaining_nodes > 2 else 0.3
    weight_destination = 1.0 - weight_current
    current_tour = [current_node] + list(unvisited_nodes) + [destination_node]
    improved = True
    while improved:
        improved = False
        for i in range(1, len(current_tour) - 2):
            for j in range(i + 1, len(current_tour) - 1):
                if j - i == 1: continue
                before = distance_matrix[current_tour[i-1]][current_tour[i]] + distance_matrix[current_tour[j]][current_tour[j+1]]
                after = distance_matrix[current_tour[i-1]][current_tour[j]] + distance_matrix[current_tour[i]][current_tour[j+1]]
                if after < before:
                    current_tour[i:j+1] = reversed(current_tour[i:j+1])
                    improved = True
    next_node = min(unvisited_nodes, key=lambda node: weight_current * distance_matrix[current_node][node] + weight_destination * distance_matrix[node][destination_node])
    return next_node
