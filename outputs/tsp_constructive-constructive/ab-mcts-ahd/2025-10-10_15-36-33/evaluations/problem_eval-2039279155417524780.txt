def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    penalty_factor = 1.1
    heuristic_weight = 0.5
    improved = True
    while improved:
        improved = False
        min_distance = float('inf')
        next_node = None
        for node in unvisited_nodes:
            if node == destination_node:
                distance = distance_matrix[current_node][node] * penalty_factor * (1 + heuristic_weight)
            else:
                distance = distance_matrix[current_node][node] - heuristic_weight * distance_matrix[node][destination_node]
            if distance < min_distance:
                min_distance = distance
                next_node = node
        heuristic_weight = heuristic_weight * 1.1 if next_node != destination_node else heuristic_weight * 0.9
        if next_node == destination_node and len(unvisited_nodes) > 1:
            improved = True
    return next_node
