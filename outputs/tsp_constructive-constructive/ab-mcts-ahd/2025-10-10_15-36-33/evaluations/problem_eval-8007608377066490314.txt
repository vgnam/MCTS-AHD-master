def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    heuristic_weight = 0.5
    improved = True
    while improved:
        improved = False
        next_node = min(unvisited_nodes, key=lambda node: distance_matrix[current_node][node] - heuristic_weight * distance_matrix[node][destination_node])
        heuristic_before = distance_matrix[current_node][next_node] - heuristic_weight * distance_matrix[next_node][destination_node]
        for node in unvisited_nodes - {next_node}:
            heuristic_after = distance_matrix[current_node][node] - heuristic_weight * distance_matrix[node][destination_node]
            if heuristic_after < heuristic_before:
                next_node = node
                heuristic_before = heuristic_after
                improved = True
        heuristic_weight = heuristic_weight * 1.1 if improved else heuristic_weight * 0.9

    if destination_node in unvisited_nodes:
        destination_heuristic = distance_matrix[current_node][destination_node] - heuristic_weight * distance_matrix[destination_node][destination_node]
        if destination_heuristic < distance_matrix[current_node][next_node] - heuristic_weight * distance_matrix[next_node][destination_node]:
            next_node = destination_node

    if len(unvisited_nodes) > 1:
        lookahead_nodes = unvisited_nodes - {next_node}
        lookahead_node = min(lookahead_nodes, key=lambda node: distance_matrix[next_node][node] + heuristic_weight * distance_matrix[node][destination_node])
        if distance_matrix[next_node][lookahead_node] + distance_matrix[lookahead_node][destination_node] < distance_matrix[next_node][destination_node]:
            next_node = lookahead_node

    return next_node
