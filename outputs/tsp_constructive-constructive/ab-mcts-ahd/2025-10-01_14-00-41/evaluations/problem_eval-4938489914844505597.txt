def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    def score(node):
        immediate_distance = distance_matrix[current_node][node]
        remaining_distances = [distance_matrix[node][unvisited] for unvisited in unvisited_nodes if unvisited != node]
        avg_remaining_distance = sum(remaining_distances) / (len(remaining_distances) + 1e-9)  # Avoid division by zero
        distance_to_destination = distance_matrix[node][destination_node]
        # Combine the scoring elements, giving higher priority to nodes that are closer to the current node, 
        # have a lower average distance to remaining nodes, and are relatively closer to the destination node.
        return immediate_distance / ((1 + avg_remaining_distance) * (1 + distance_to_destination))
    
    next_node = min(unvisited_nodes, key=score)
    return next_node
