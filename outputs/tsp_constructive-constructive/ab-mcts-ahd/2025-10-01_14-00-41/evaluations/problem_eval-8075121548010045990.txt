import random

def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    def score(node):
        direct_dist = distance_matrix[current_node][node]
        dest_dist = distance_matrix[node][destination_node]
        inv_dest_dist = 1 / (1 + dest_dist)
        if len(unvisited_nodes) > 1:
            avg_distance_to_unvisited = sum(distance_matrix[node][unvisited_node] for unvisited_node in unvisited_nodes if unvisited_node != node) / (len(unvisited_nodes) - 1)
            clustering_factor = sum(1 for unvisited_node in unvisited_nodes if unvisited_node != node and distance_matrix[node][unvisited_node] < avg_distance_to_unvisited) / (len(unvisited_nodes) - 1)
        else:
            avg_distance_to_unvisited = 0
            clustering_factor = 0
        return direct_dist + inv_dest_dist + 0.2 * (1 / (1 + avg_distance_to_unvisited)) + 0.1 * clustering_factor

    scores = {node: score(node) for node in unvisited_nodes}
    min_score = min(scores.values())
    max_score = max(scores.values())
    score_range = max_score - min_score
    if score_range == 0:
        return random.choice(list(unvisited_nodes))
    probabilities = {node: (max_score - score) / score_range for node, score in scores.items()}
    probabilities = {node: prob / sum(probabilities.values()) for node, prob in probabilities.items()}
    next_node = random.choices(list(probabilities.keys()), weights=probabilities.values(), k=1)[0]
    return next_node
