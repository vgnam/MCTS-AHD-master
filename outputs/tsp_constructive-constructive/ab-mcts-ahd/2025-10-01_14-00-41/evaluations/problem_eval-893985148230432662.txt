def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node
    
    def score(node):
        distance_to_current = distance_matrix[current_node][node]
        connectivity_score = sum(distance_matrix[node][unvisited_node] for unvisited_node in unvisited_nodes if unvisited_node != node)
        return distance_to_current * (1 + (connectivity_score / (len(unvisited_nodes) - 1) if len(unvisited_nodes) > 1 else 0))
    
    next_node = min(unvisited_nodes, key=score)
    return next_node
