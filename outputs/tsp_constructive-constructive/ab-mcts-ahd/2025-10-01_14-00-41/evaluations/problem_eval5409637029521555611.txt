import numpy as np

def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    def score(node):
        return distance_matrix[current_node][node] * (1 + distance_matrix[node][destination_node] / np.max(distance_matrix))

    scores = np.array([score(node) for node in unvisited_nodes])
    temperature = 0.5  # Can be adjusted for different levels of exploration/exploitation
    probabilities = np.exp(-scores / temperature) / np.sum(np.exp(-scores / temperature))
    next_node = np.random.choice(list(unvisited_nodes), p=probabilities)
    return next_node
