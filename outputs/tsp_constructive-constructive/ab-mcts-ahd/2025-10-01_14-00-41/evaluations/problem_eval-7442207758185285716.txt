def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    def score(node):
        avg_distance_to_unvisited = sum(distance_matrix[node][unvisited_node] for unvisited_node in unvisited_nodes if unvisited_node != node) / (len(unvisited_nodes) - 1) if len(unvisited_nodes) > 1 else 0
        weight = max(1, len(unvisited_nodes) / len(distance_matrix))
        return distance_matrix[current_node][node] * (1 + avg_distance_to_unvisited / weight) / (1 + distance_matrix[node][destination_node])

    next_node = min(unvisited_nodes, key=score)
    return next_node
