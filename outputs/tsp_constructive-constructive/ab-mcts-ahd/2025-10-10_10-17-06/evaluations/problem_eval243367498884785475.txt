def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    total_unvisited = len(unvisited_nodes)
    total_nodes = len(distance_matrix)
    proximity_weight = 0.6 + 0.2 * (total_unvisited / (total_nodes + total_unvisited))
    lookahead_weight = 0.2
    destination_weight = 0.2 - 0.1 * (total_unvisited / (total_nodes + total_unvisited))
    diversity_weight = 0.2

    def calculate_weighted_cost(node):
        proximity = distance_matrix[current_node][node]
        lookahead = sum(1 for n in unvisited_nodes if n != node and distance_matrix[node][n] < 1.5 * proximity)
        destination_proximity = distance_matrix[node][destination_node]
        diversity = len([n for n in unvisited_nodes if distance_matrix[node][n] < 2 * proximity])
        return (proximity_weight * proximity) + (lookahead_weight * lookahead) + (destination_weight * destination_proximity) + (diversity_weight * diversity)

    next_node = min(unvisited_nodes, key=lambda node: calculate_weighted_cost(node))
    return next_node
