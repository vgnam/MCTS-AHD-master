def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    remaining_nodes = len(unvisited_nodes)
    if remaining_nodes == 0:
        return destination_node

    # Calculate centrality of each node (average distance to all other nodes)
    centrality = {}
    for node in unvisited_nodes:
        avg_distance = sum(distance_matrix[node][other] for other in unvisited_nodes) / remaining_nodes
        centrality[node] = avg_distance

    # Dynamic weights for proximity and centrality
    proximity_weight = 0.6 + 0.4 * (remaining_nodes / 50)  # Higher weight when more nodes remain
    centrality_weight = 0.4 * (1 - remaining_nodes / 50)  # Higher weight as fewer nodes remain

    # Normalize centrality values
    max_centrality = max(centrality.values()) if centrality else 1
    min_centrality = min(centrality.values()) if centrality else 0
    normalized_centrality = {node: (centrality[node] - min_centrality) / (max_centrality - min_centrality + 1e-6) for node in centrality}

    # Score nodes based on weighted combination
    def score(node):
        proximity_score = distance_matrix[current_node][node] / (1 + distance_matrix[node][destination_node])
        centrality_score = normalized_centrality.get(node, 0)
        return proximity_weight * proximity_score + centrality_weight * centrality_score

    next_node = min(unvisited_nodes, key=score)
    return next_node
