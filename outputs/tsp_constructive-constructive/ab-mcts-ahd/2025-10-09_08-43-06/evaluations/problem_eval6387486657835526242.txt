import random
import math

def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return None

    # Calculate weights for nodes based on distance and destination bias
    total_unvisited = len(unvisited_nodes)
    destination_bias = math.exp(-0.5 * total_unvisited)  # Exponentially decreasing bias as nodes are visited

    weights = []
    for node in unvisited_nodes:
        distance = distance_matrix[current_node][node]
        if node == destination_node:
            weight = destination_bias * (1.0 / (distance + 1e-6))  # Avoid division by zero
        else:
            weight = (1.0 / (distance + 1e-6))  # Inverse distance weighting
        weights.append(weight)

    # Normalize weights to form a probability distribution
    total_weight = sum(weights)
    probabilities = [w / total_weight for w in weights]

    # Select next node based on weighted probabilities
    next_node = random.choices(unvisited_nodes, weights=probabilities, k=1)[0]
    return next_node
