def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    remaining_nodes = len(unvisited_nodes)
    total_nodes = len(distance_matrix)
    dynamic_weight_proximity = 0.5 + (total_nodes - remaining_nodes) * 0.05
    dynamic_weight_destination = 1 - dynamic_weight_proximity
    recent_nodes = set()  # To keep track of recently visited nodes for penalty
    if total_nodes - remaining_nodes > 1:
        recent_nodes.add(current_node)
    penalty_factor = 0.1 * len(recent_nodes)
    
    next_node = min(unvisited_nodes, key=lambda node: 
        (distance_matrix[current_node][node] * dynamic_weight_proximity) +
        (distance_matrix[node][destination_node] * dynamic_weight_destination) +
        (penalty_factor if node in recent_nodes else 0))
    return next_node
