def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    remaining_nodes = len(unvisited_nodes)
    total_nodes = len(distance_matrix)
    dynamic_weight = 1.0 + (remaining_nodes / total_nodes) * 0.2

    next_node = min(unvisited_nodes, key=lambda node:
        (distance_matrix[current_node][node] * dynamic_weight * 0.7) +
        (distance_matrix[node][destination_node] * (1 - dynamic_weight) * 0.2) +
        ((distance_matrix[node][destination_node] / sum(distance_matrix[node][unvisited] for unvisited in unvisited_nodes)) * (1 - dynamic_weight) * 0.1))

    return next_node
