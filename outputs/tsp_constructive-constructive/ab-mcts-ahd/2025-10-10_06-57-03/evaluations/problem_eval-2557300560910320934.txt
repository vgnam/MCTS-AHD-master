def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    def score(node):
        immediate_cost = distance_matrix[current_node][node]
        destination_attraction = (1 / (1 + distance_matrix[node][destination_node])) ** 2
        remaining_nodes = len(unvisited_nodes)
        exploration_factor = sum(distance_matrix[node][n] for n in unvisited_nodes) / remaining_nodes if remaining_nodes else 0
        dynamic_weight_immediate_cost = immediate_cost * (1 + 0.1 * (1 - remaining_nodes / total_nodes))
        dynamic_weight_destination_attraction = destination_attraction * (1 + 0.2 * remaining_nodes / total_nodes)
        alpha = 0.5 + 0.5 * (remaining_nodes / len(distance_matrix)) ** 1.5
        return alpha * dynamic_weight_immediate_cost + (1 - alpha) * dynamic_weight_destination_attraction - 0.5 * exploration_factor

    total_nodes = len(distance_matrix)
    next_node = min(unvisited_nodes, key=score)
    return next_node
