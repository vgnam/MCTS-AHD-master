def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node
    remaining_nodes = len(unvisited_nodes)
    alpha = 0.5 + 0.5 * (remaining_nodes / len(distance_matrix)) ** 1.5
    center_node = min(unvisited_nodes, key=lambda n: sum(distance_matrix[n][m] for m in unvisited_nodes))
    exploration_factor = sum(distance_matrix[node][center_node] for node in unvisited_nodes) / len(unvisited_nodes) if unvisited_nodes else 0
    penalty = 0.4 if remaining_nodes > 2 else 0.6
    next_node = min(unvisited_nodes, key=lambda node: alpha * distance_matrix[current_node][node] + (1 - alpha) * distance_matrix[node][destination_node] - penalty * exploration_factor)
    return next_node
