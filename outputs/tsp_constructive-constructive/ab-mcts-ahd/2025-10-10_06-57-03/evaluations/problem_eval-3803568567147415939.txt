def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    def score(node):
        immediate_cost = distance_matrix[current_node][node]
        destination_attraction = 1.0 / (1 + distance_matrix[node][destination_node])
        remaining_nodes = len(unvisited_nodes)
        dynamic_weight_immediate_cost = immediate_cost * (1 + 0.1 * (1 - remaining_nodes / total_nodes))
        dynamic_weight_destination_attraction = destination_attraction * (1 + 0.1 * remaining_nodes / total_nodes)
        return dynamic_weight_immediate_cost * dynamic_weight_destination_attraction
    total_nodes = len(distance_matrix)
    next_node = min(unvisited_nodes, key=score)
    return next_node
