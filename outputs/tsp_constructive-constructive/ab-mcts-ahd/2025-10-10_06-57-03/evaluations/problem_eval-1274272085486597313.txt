def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    remaining = len(unvisited_nodes)
    if remaining == 0:
        return destination_node
    weight_current = 1 / (remaining ** 2 + 1)
    weight_destination = remaining ** 2 / (remaining ** 2 + 1)
    exploration_penalty = 0.4 if remaining > 2 else 0.6
    def score(node):
        immediate_cost = distance_matrix[current_node][node]
        destination_attraction = weight_destination * distance_matrix[node][destination_node]
        exploration_factor = sum(distance_matrix[node][n] for n in unvisited_nodes) / len(unvisited_nodes) if unvisited_nodes else 0
        return immediate_cost + destination_attraction - exploration_penalty * exploration_factor
    next_node = min(unvisited_nodes, key=score)
    return next_node
