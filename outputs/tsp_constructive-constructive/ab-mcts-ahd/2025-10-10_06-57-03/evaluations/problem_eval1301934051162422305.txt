def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    next_node = min(unvisited_nodes, key=lambda node: distance_matrix[current_node][node] - 0.5 * distance_matrix[node][destination_node] + 0.2 * (sum(distance_matrix[node][n] for n in unvisited_nodes) / len(unvisited_nodes) if unvisited_nodes else 0))
    return next_node
