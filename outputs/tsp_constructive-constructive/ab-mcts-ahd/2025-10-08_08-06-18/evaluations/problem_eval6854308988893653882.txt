def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    next_node = None
    best_score = float('inf')

    for node in unvisited_nodes:
        current_to_node = distance_matrix[current_node][node]
        total_remaining = sum(distance_matrix[node][other] for other in unvisited_nodes if other != node)
        score = current_to_node + total_remaining
        if score < best_score:
            best_score = score
            next_node = node

    return next_node
