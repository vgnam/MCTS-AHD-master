import math

def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node
    num_unvisited = len(unvisited_nodes)
    proximity_weight = 0.8 * math.exp(-num_unvisited / len(distance_matrix))
    next_node = min(unvisited_nodes, key=lambda node: proximity_weight * distance_matrix[current_node][node] + (1 - proximity_weight) * distance_matrix[node][destination_node])
    return next_node
