def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    def neighbor_count(node):
        return len([n for n in unvisited_nodes if distance_matrix[node][n] != float('inf')])

    next_node = min(
        unvisited_nodes,
        key=lambda node: distance_matrix[current_node][node] + 
                         0.5 * sum((distance_matrix[node][n] - sum(distance_matrix[node][m] for m in unvisited_nodes - {node}) / len(unvisited_nodes - {node}))**2 for n in unvisited_nodes - {node}) / len(unvisited_nodes - {node}) if (unvisited_nodes - {node}) else 0 -
                         0.2 * neighbor_count(node)
    )
    return next_node
