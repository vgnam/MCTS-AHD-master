def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not hasattr(select_next_node, 'iteration'):
        select_next_node.iteration = 0
    iteration = select_next_node.iteration

    def node_score(node):
        proximity = distance_matrix[current_node][node]
        progress_towards_destination = distance_matrix[node][destination_node]
        weight_proximity = 0.8 - (0.05 * iteration)
        weight_progress = 0.6 + (0.03 * iteration)
        return (weight_proximity * proximity) + (weight_progress / (progress_towards_destination + 1))

    next_node = min(unvisited_nodes, key=node_score)
    select_next_node.iteration += 1
    return next_node
