def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    exploration_factor = 1.0 / (len(unvisited_nodes) + 1)
    weight_proximity = 1.2 + exploration_factor
    weight_progress = 1.0 + exploration_factor
    next_node = min(unvisited_nodes, key=lambda node: (weight_proximity * distance_matrix[current_node][node]) + (weight_progress / (distance_matrix[node][destination_node] + 1)))
    return next_node
