def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    def polygon_perimeter(current, node, next_node):
        return (distance_matrix[current][node] + 
                distance_matrix[node][next_node] + 
                distance_matrix[next_node][current])

    total_nodes = len(distance_matrix)
    remaining_nodes = len(unvisited_nodes)
    weight_immediate = 1 - remaining_nodes / total_nodes
    weight_polygon = remaining_nodes / total_nodes
    
    candidates = []
    for node in unvisited_nodes:
        immediate_distance = distance_matrix[current_node][node]
        remaining_unvisited = unvisited_nodes - {node}
        if remaining_unvisited:
            next_nearest = min(remaining_unvisited, key=lambda n: distance_matrix[node][n])
            perimeter = polygon_perimeter(current_node, node, next_nearest)
            total_distance = weight_immediate * immediate_distance + weight_polygon * perimeter
        else:
            total_distance = immediate_distance + distance_matrix[node][destination_node]
        candidates.append((node, total_distance))

    next_node, _ = min(candidates, key=lambda x: x[1])
    return next_node
