def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    def candidate_score(node):
        distance_to_node = distance_matrix[current_node][node]
        distance_to_destination = distance_matrix[node][destination_node]
        remaining_unvisited = unvisited_nodes - {node}
        remaining_connections = sum(distance_matrix[node][n] for n in remaining_unvisited) if remaining_unvisited else 0
        total_nodes = len(distance_matrix)
        remaining_nodes = len(unvisited_nodes)
        weight_immediate = 1 - remaining_nodes / total_nodes
        weight_destination = remaining_nodes / total_nodes
        weight_remaining = 1 - weight_destination
        return weight_immediate * distance_to_node + weight_destination * (1.0 / (1.0 + distance_to_destination)) ** 2 + weight_remaining * (1 / (remaining_connections + 1e-6))
    next_node = min(unvisited_nodes, key=lambda node: candidate_score(node))
    return next_node
