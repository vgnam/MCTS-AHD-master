def select_next_node(current_node, destination_node, unvisited_nodes, distance_matrix):
    if not unvisited_nodes:
        return destination_node

    candidates = []
    for node in unvisited_nodes:
        immediate_distance = distance_matrix[current_node][node]
        remaining_unvisited = unvisited_nodes - {node}
        if remaining_unvisited:
            avg_future_distance = sum(distance_matrix[node][n] for n in remaining_unvisited) / len(remaining_unvisited)
            shortest_future_distance = min(distance_matrix[node][n] for n in remaining_unvisited)
            total_distance = (immediate_distance * (1 + len(remaining_unvisited)) + avg_future_distance + shortest_future_distance) / (3 + len(remaining_unvisited))
        else:
            total_distance = immediate_distance + distance_matrix[node][destination_node]
        candidates.append((node, total_distance))

    next_node, _ = min(candidates, key=lambda x: x[1])
    return next_node
